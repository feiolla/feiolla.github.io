<script>
(function(){
  /* ==================== 电路参数 (固定) ==================== */
  const VCC = 12, RC = 2, RL = 2;
  const Rac = (RC*RL)/(RC+RL);
  const ibCurves = [10,20,30,40,50,60,70,80,90];

  const VT = 0.026;

  const VCE_SAT = 0.3;
  const IC_MIN  = 0.02;
  const IC_MAX  = (VCC - VCE_SAT) / RC;

  const Y_MAX = 9;
  const Y_TICK_STEP = 1.5;
  const EARLY_V = 80;

  function V_knee(ib_uA){
    return 0.3 + 0.001 * ib_uA;
  }
  function iC_at_knee(ib_uA){
    const ibmA = ib_uA / 1000;
    const vk = V_knee(ib_uA);
    return 100 * ibmA * (1 - Math.exp(-vk/0.3)) * (1 + vk/EARLY_V);
  }

  const MACARON = {
    undistorted: '#A8E6CF',
    cutoff:      '#B5D8FF',
    sat:         '#FFB3BA',
    severe:      '#D4B5FF',
    dcLine:      '#C9C9C9',
    acLine:      '#FFDAC1',
    qPoint:      '#FFC8A2',
    kneeCurve:   '#C8A8E9',
    kneeDot:     '#9B7BC8',
    kneeDashed:  '#C8C0D8',
    refLine:     '#B0A098',
    grid:        '#F5EAE4',
    axis:        '#A8958A'
  };

  /* ==================== 常温设计 ==================== */
  const ICQ_0  = (VCC - VCE_SAT) / (RC + Rac);
  const VCEQ_0 = VCC - ICQ_0 * RC;
  const VCC_0  = VCEQ_0 + ICQ_0 * Rac;

  const VOM_MAX_UP   = ICQ_0 * Rac;
  const VOM_MAX_DOWN = VCEQ_0 - VCE_SAT;
  const VOM_MAX = Math.min(VOM_MAX_UP, VOM_MAX_DOWN);

  const AV_0 = 60;
  const VI_AMPLITUDE = 0.052;

  /* ==================== 温度模型 ==================== */
  const BETA_0 = 100;
  function icqFromTemp(T){
    const dT = T - 25;
    let k = (dT >= 0) ? 0.0116 : 0.1030;
    let icq = ICQ_0 * Math.exp(k * dT);
    if (icq > IC_MAX) icq = IC_MAX;
    if (icq < 0) icq = 0;
    return icq;
  }

  /* ==================== 主画布 ==================== */
  const canvas = document.getElementById('bjtCanvas');
  const ctx = canvas.getContext('2d');
  const M = {l:90, r:55, t:38, b:72};
  const PW = canvas.width - M.l - M.r;
  const PH = canvas.height - M.t - M.b;
  const X0=0, X1=12, Y0=0, Y1=Y_MAX;
  const cX = v => M.l + (v-X0)/(X1-X0)*PW;
  const cY = i => M.t + PH - (i-Y0)/(Y1-Y0)*PH;

  const inCv = document.getElementById('inputWave');
  const inCtx = inCv.getContext('2d');
  const outCv = document.getElementById('outputWave');
  const outCtx = outCv.getContext('2d');

  let temperature = 25;
  let icQ = ICQ_0;
  let sweepTimer = null;
  let sweepDir = 1;

  function getQ(){
    let vce = VCC - icQ*RC;
    let ib = (icQ/BETA_0)*1000;
    if (vce < 0.5) ib = ib*(1+(0.5-vce)*2);
    return {ic: icQ, vce, ib};
  }

  function getGain(){
    const gm = ICQ_0 / VT;
    return {gm, Av: AV_0};
  }

  function findSatCrossPoint(q){
    const qIc = q.ic, qVce = q.vce;
    const vcc0 = qVce + qIc * Rac;

    const N = 400;
    let prevInside = null;
    let crossVce = null, crossIc = null;

    for (let i=0; i<=N; i++){
      const ic = (i/N) * (vcc0 / Rac);
      const vce = vcc0 - ic * Rac;

      let inside = false;
      if (vce < 0.3) {
        inside = true;
      } else {
        const ibAtVce = (vce - 0.3) / 0.001;
        if (ibAtVce >= 0 && ibAtVce <= 300) {
          const icKnee = iC_at_knee(ibAtVce);
          inside = (ic >= icKnee);
        } else if (ibAtVce > 300) {
          inside = false;
        }
      }

      if (prevInside === false && inside === true){
        crossVce = vce;
        crossIc = ic;
        break;
      }
      prevInside = inside;
    }

    return {vce: crossVce, ic: crossIc, vcc0};
  }

  function computeDynamicEnds(q, deltaIc){
    const qIc = q.ic, qVce = q.vce;
    const vcc0 = qVce + qIc * Rac;

    const cross = findSatCrossPoint(q);

    const icTop_ideal = qIc + deltaIc;
    const satClipped = (cross.ic !== null) && (icTop_ideal >= cross.ic);
    let icTop, vceTop;
    if (satClipped) {
      icTop = cross.ic;
      vceTop = cross.vce;
    } else {
      icTop = icTop_ideal;
      vceTop = qVce - deltaIc * Rac;
    }

    const icBot_ideal = qIc - deltaIc;
    const cutClipped = (icBot_ideal <= 0);
    let icBot, vceBot;
    if (cutClipped) {
      icBot = 0;
      vceBot = vcc0;
    } else {
      icBot = icBot_ideal;
      vceBot = qVce + deltaIc * Rac;
    }

    return {
      icTop, vceTop, satClipped,
      icBot, vceBot, cutClipped,
      icTop_ideal, icBot_ideal,
      vcc0,
      cross_vce: cross.vce,
      cross_ic: cross.ic
    };
  }

  function drawCurves(){
    for (let ib of ibCurves){
      const ibmA = ib/1000;
      ctx.beginPath();
      let first = true;
      for (let i=0;i<=240;i++){
        const vce = X0 + (i/240)*(X1-X0);
        let ic = BETA_0*ibmA*(1-Math.exp(-vce/0.3))*(1+vce/EARLY_V);
        if (ic > Y1 + 0.4) continue;
        const px = cX(vce), py = cY(ic);
        if (first){ ctx.moveTo(px,py); first=false; }
        else ctx.lineTo(px,py);
      }
      ctx.strokeStyle='#A8C5D8'; ctx.lineWidth=1.7; ctx.globalAlpha=0.7;
      ctx.stroke();

      const lv = 11.3;

  function findCriticalSatCurve(q, dynEnds){
    if (!dynEnds.satClipped) return null;

    let bestIb = null, bestDist = Infinity;
    for (let ib of ibCurves){
        const ibmA = ib / 1000;
        const icOnCurve = BETA_0 * ibmA * (1 - Math.exp(-VCE_SAT/0.3)) * (1 + VCE_SAT/EARLY_V);
        const dist = Math.abs(icOnCurve - dynEnds.cross_ic);
        if (dist < bestDist){
            bestDist = dist;
            bestIb = ib;
        }
    }
    if (bestIb === null) return null;
    return {ib: bestIb};
  }

  function drawOutputWave(q, dynEnds){
    const w=outCv.width, h=outCv.height;
    outCtx.clearRect(0,0,w,h);
    outCtx.fillStyle='#FFFCF9'; outCtx.fillRect(0,0,w,h);

    outCtx.strokeStyle='#F5EAE4'; outCtx.lineWidth=0.7;
    for (let i=1;i<6;i++){
      const x=(i/6)*w; outCtx.beginPath(); outCtx.moveTo(x,0); outCtx.lineTo(x,h); outCtx.stroke();
    }

    const qVce = q.vce;
    const { Av } = getGain();
    const VO_AMPLITUDE = VI_AMPLITUDE * Av;

    const VCE_CUTOFF = dynEnds.vcc0;
    const VCE_SAT_BOUND = VCE_SAT;

    const N = 240;
    const idealVce = [];
    const actualVce = [];
    let clipTop = false;
    let clipBot = false;

    for (let i=0;i<=N;i++){
      const t = i/N;
      const angle = t * Math.PI * 4;
      const vi = VI_AMPLITUDE * Math.sin(angle);
      const vo = -Av * vi;
      const ideal = qVce + vo;
      idealVce.push(ideal);

      let a = ideal;
      if (a > VCE_CUTOFF) { a = VCE_CUTOFF; clipTop = true; }
      if (a < VCE_SAT_BOUND) { a = VCE_SAT_BOUND; clipBot = true; }
      actualVce.push(a);
    }

    const vLo = 0.0, vHi = 12.0;
    const padTop = 40;
    const padBot = 30;
    const mapY = v => {
      const t = (v - vLo) / (vHi - vLo);
      return (h - padBot) - t * (h - padTop - padBot);
    };

    outCtx.beginPath();
    outCtx.moveTo(0, mapY(VCE_CUTOFF)); outCtx.lineTo(w, mapY(VCE_CUTOFF));
    outCtx.strokeStyle = 'rgba(107,168,224,0.75)';
    outCtx.lineWidth = 1.6; outCtx.setLineDash([4,4]); outCtx.stroke(); outCtx.setLineDash([]);
    outCtx.font='13px Segoe UI,sans-serif'; outCtx.fillStyle='#4A88C0';
    outCtx.fillText('截止边界 (iC=0)', w - 130, mapY(VCE_CUTOFF) - 6);

    outCtx.beginPath();
    outCtx.moveTo(0, mapY(VCE_SAT_BOUND)); outCtx.lineTo(w, mapY(VCE_SAT_BOUND));
    outCtx.strokeStyle = 'rgba(224,122,139,0.75)';
    outCtx.lineWidth = 1.6; outCtx.setLineDash([4,4]); outCtx.stroke(); outCtx.setLineDash([]);
    outCtx.font='13px Segoe UI,sans-serif'; outCtx.fillStyle='#D06078';
    outCtx.fillText('饱和边界 0.3V', w - 118, mapY(VCE_SAT_BOUND) + 18);

    outCtx.beginPath();
    for (let i=0;i<=N;i++){
      const x = (i/N)*w;
      const y = mapY(idealVce[i]);
      if (i===0) outCtx.moveTo(x,y); else outCtx.lineTo(x,y);
    }
    outCtx.strokeStyle='rgba(200,190,185,0.65)';
    outCtx.lineWidth=1.3; outCtx.setLineDash([3,3]); outCtx.stroke(); outCtx.setLineDash([]);

    outCtx.beginPath();
    for (let i=0;i<=N;i++){
      const x = (i/N)*w;
      const y = mapY(actualVce[i]);
      if (i===0) outCtx.moveTo(x,y); else outCtx.lineTo(x,y);
    }
    let waveColor = '#7ECFC0';
    if (clipTop && clipBot) waveColor = '#B58AE0';
    else if (clipBot) waveColor = '#FF8FA0';
    else if (clipTop) waveColor = '#7EB8E8';
    outCtx.strokeStyle=waveColor; outCtx.lineWidth=3.2; outCtx.stroke();

    if (clipTop){
      outCtx.fillStyle='rgba(181,216,255,0.4)';
      outCtx.fillRect(0, mapY(VCE_CUTOFF)-4, w, 9);
      outCtx.font='bold 14px Segoe UI,sans-serif'; outCtx.fillStyle='#4A88C0';
      outCtx.fillText('▲ 截止削顶', 10, mapY(VCE_CUTOFF) - 12);
    }
    if (clipBot){
      outCtx.fillStyle='rgba(255,179,186,0.4)';
      outCtx.fillRect(0, mapY(VCE_SAT_BOUND)-4, w, 9);
      outCtx.font='bold 14px Segoe UI,sans-serif'; outCtx.fillStyle='#D06078';
      outCtx.fillText('▼ 饱和削底', 10, mapY(VCE_SAT_BOUND) + 24);
    }

    const qY = mapY(qVce);
    outCtx.beginPath(); outCtx.moveTo(0,qY); outCtx.lineTo(w,qY);
    outCtx.strokeStyle='#FFC8A2'; outCtx.lineWidth=1.6;
    outCtx.setLineDash([4,4]); outCtx.stroke(); outCtx.setLineDash([]);
    outCtx.font='bold 14px Segoe UI,sans-serif'; outCtx.fillStyle='#C08860';
    outCtx.fillText('Q', 10, qY-6);

    return {clipTop, clipBot, voAmp: VO_AMPLITUDE};
  }

      msg.innerHTML=`⚠️ <b>截止失真</b>：i<sub>B</sub>≈0 使 i<sub>C</sub>≈0，动态最低端停在 i<sub>C</sub>=0 轴 → <b>输出波形顶部被削平</b>`;
      msg.className='msg bad'; clipTag.textContent='顶部削平';
      isBad=true;
    } else {
      const vceTop = dynEnds.vceTop;
      const icBot = dynEnds.icBot;
      const headroomTop = vceTop - VCE_SAT;
      const headroomBot = icBot;

      if (headroomTop < 0.3 && headroomBot < 0.3){
        badge.textContent='接近双边界'; badge.className='badge warn';
        msg.innerHTML=`⚠️ Q点同时靠近截止与饱和边界`;
        msg.className='msg warn'; clipTag.textContent='临界';
        isWarn=true;
      } else if (headroomTop < 0.3){
        badge.textContent='接近饱和'; badge.className='badge warn';
        msg.innerHTML=`⚠️ 动态最高端接近饱和边界轨迹，温度继续升高 → 底部将被削平`;
        msg.className='msg warn'; clipTag.textContent='接近饱和';
        isWarn=true;
      } else if (headroomBot < 0.3){
        badge.textContent='接近截止'; badge.className='badge warn';
        msg.innerHTML=`⚠️ 动态最低端接近 i<sub>C</sub>=0，温度继续降低 → 顶部将被削平`;
        msg.className='msg warn'; clipTag.textContent='接近截止';
        isWarn=true;
      } else {
        badge.textContent='放大区'; badge.className='badge good';
        msg.innerHTML = `✅ 线性放大正常，输出 ±${(clip.voAmp).toFixed(2)}V`;
        msg.className='msg good'; clipTag.textContent='无削波';
      }
    }

    if (critSat && !isBad) {
      const baseMsg = msg.innerHTML;
      msg.innerHTML = baseMsg + `<br><span style="color:#8E6DB5; font-size:0.9rem;">⚠️ 动态最高端已达 <b>临界饱和曲线 iB=${critSat.ib}μA</b>，接近拐点</span>`;
      if (!isWarn && !isBad) {
        msg.className = 'msg warn';
        badge.textContent = '接近饱和';
        badge.className = 'badge warn';
        clipTag.textContent = '临界饱和';
      }
    }

    outCv.style.border = isBad ? '2px solid #FFB3BA' :
                          isWarn ? '2px solid #FFE2A8' :
                                   '2px solid #A8E6CF';

    let concl = '';
    const icq = q.ic;
    if (temperature > 45) {
      concl = `<b>🔥 高温 (${temperature.toFixed(0)}°C)：</b>
               i<sub>CQ</sub> = <span class="kv">${icq.toFixed(2)} mA</span> ↑
               → Q 点沿<b>直流负载线</b>下移 → 动态最高端撞到 <b>饱和边界轨迹</b>（拐点连线）与交流负载线的交点 →
               i<sub>C</sub> 被钳在此交点（<b>饱和限流</b>）→ <b>输出波形底部削平（饱和失真）</b>`;
    } else if (temperature < 10) {
      concl = `<b>❄️ 低温 (${temperature.toFixed(0)}°C)：</b>
               i<sub>CQ</sub> = <span class="kv">${icq.toFixed(2)} mA</span> ↓
               → Q 点沿<b>直流负载线</b>上移 → 动态最低端 i<sub>B</sub>≈0 使 i<sub>C</sub>≈0，被钳在 i<sub>C</sub>=0 轴 →
               <b>输出波形顶部削平（截止失真）</b>`;
    } else {
      concl = `<b>🌤️ 常温 (${temperature.toFixed(0)}°C)：</b>
               Q 点位于交流负载线对称中心 <b>v<sub>CEQ</sub>=${VCEQ_0.toFixed(2)}V, i<sub>CQ</sub>=${ICQ_0.toFixed(2)}mA</b>，
               最大不失真输出 ±${VOM_MAX.toFixed(2)}V；当前输入 52mV 对应输出 ±${(clip.voAmp).toFixed(2)}V，<b>线性放大状态稳定</b>。`;
    }
    document.getElementById('conclusionText').innerHTML = concl + 
      '<br><br><b>→ 固定偏置电路无法稳定 Q 点，温度漂移必然使动态工作点轨迹撞边界，必须引入分压式偏置 + 射极电阻等负反馈措施。</b>';
  }

  function draw(){
    ctx.clearRect(0,0,canvas.width,canvas.height);
    drawAxes();
    drawRefVce03();
    drawCurves();
    drawKneeCurve();

    const q = getQ();
    const {Av} = getGain();
    const deltaIc = (VI_AMPLITUDE * Av) / RC;

    const dynEnds = computeDynamicEnds(q, deltaIc);

    const critSat = findCriticalSatCurve(q, dynEnds);
    if (critSat) drawCriticalSatCurve(critSat);

    drawDC();
    drawAC(q);
    drawDynamicTrace(q, dynEnds);
    drawQ(q);

    drawInputWave();
    const clip = drawOutputWave(q, dynEnds);
    updateUI(q, clip, critSat, dynEnds);
  }

  function setTemperature(T){
    temperature = Math.max(0, Math.min(70, T));
    icQ = icqFromTemp(temperature);
    document.getElementById('tempSlider').value = temperature;
    draw();
  }

  const tempSlider = document.getElementById('tempSlider');
  tempSlider.addEventListener('input', e => {
    stopSweep();
    setTemperature(parseFloat(e.target.value));
  });

  document.getElementById('btnCold').addEventListener('click', () => {
    stopSweep();
    setTemperature(temperature - 10);
  });

  document.getElementById('btnHot').addEventListener('click', () => {
    stopSweep();
    setTemperature(temperature + 10);
  });

  document.getElementById('resetBtn').addEventListener('click', () => {
    stopSweep();
    setTemperature(25);
  });

  function stopSweep(){
    if (sweepTimer){
      clearInterval(sweepTimer);
      sweepTimer = null;
      document.getElementById('btnSweep').textContent = '▶ 自动扫描温度';
    }
  }

  document.getElementById('btnSweep').addEventListener('click', () => {
    if (sweepTimer){
      stopSweep();
      return;
    }
    document.getElementById('btnSweep').textContent = '⏸ 暂停扫描';
    sweepTimer = setInterval(() => {
      temperature += sweepDir * 0.6;
      if (temperature > 70){ temperature = 70; sweepDir = -1; }
      if (temperature < 0){ temperature = 0; sweepDir = 1; }
      icQ = icqFromTemp(temperature);
      document.getElementById('tempSlider').value = temperature;
      draw();
    }, 60);
  });

  setTemperature(25);
})();
